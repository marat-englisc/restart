> [!abstract] О шпаргалке
> Синтаксис встроенного языка на одной странице: конструкции, коллекции, работа с объектами и запросами. Подробности — по ссылкам на темы курса.

## Конструкции

```1C
// Условие
Если Условие Тогда
	...
ИначеЕсли ДругоеУсловие Тогда
	...
Иначе
	...
КонецЕсли;
Значение = ?(Условие, ЕслиДа, ЕслиНет);

// Циклы
Для Счетчик = 1 По 10 Цикл ... КонецЦикла;
Для Каждого Элемент Из Коллекция Цикл ... КонецЦикла;
Пока Условие Цикл ... КонецЦикла;
Прервать; Продолжить;

// Методы
Процедура Имя(Параметр1, Знач Параметр2 = 0) Экспорт ... КонецПроцедуры
Функция Имя(Параметр) Экспорт ... Возврат Результат; КонецФункции

// Ошибки
Попытка
	...
Исключение
	ЗаписьЖурналаРегистрации("Событие", УровеньЖурналаРегистрации.Ошибка, , ,
		ПодробноеПредставлениеОшибки(ИнформацияОбОшибке()));
	ВызватьИсключение;
КонецПопытки;

// Асинхронность (клиент)
Асинх Процедура Имя()
	Ответ = Ждать ВопросАсинх("Продолжить?", РежимДиалогаВопрос.ДаНет);
КонецПроцедуры
```

## Типы и проверки

| Задача | Код |
| --- | --- |
| Тип значения | `ТипЗнч(З) = Тип("Строка")` |
| Заполнено ли | `ЗначениеЗаполнено(З)` |
| Пустая строка (с пробелами) | `ПустаяСтрока(С)` |
| Пустая ссылка | `Ссылка.Пустая()` |
| Привести тип | `Число(С)`, `Строка(Ч)`, `Дата(2026, 10, 19)`, `Булево(1)` |
| Формат вывода | `Формат(Сумма, "ЧДЦ=2")`, `Формат(Дата, "ДФ=dd.MM.yyyy")` |
| Шаблон строки | `СтрШаблон("%1 шт. по %2", Кол, Цена)` |

## Коллекции

```1C
// Массив
М = Новый Массив; М.Добавить(1); М.Количество(); М[0]; М.Найти(1); М.Удалить(0);

// Структура
С = Новый Структура("Количество,Цена", 2, 150); С.Цена; С.Свойство("Цена"); С.Вставить("Сумма", 300);

// Соответствие
Соотв = Новый Соответствие; Соотв.Вставить(Товар, 100); Соотв.Получить(Товар);

// ТаблицаЗначений (сервер)
ТЗ = Новый ТаблицаЗначений;
ТЗ.Колонки.Добавить("Товар"); ТЗ.Колонки.Добавить("Сумма", Новый ОписаниеТипов("Число"));
Стр = ТЗ.Добавить(); Стр.Сумма = 100;
ТЗ.Итог("Сумма"); ТЗ.Свернуть("Товар", "Сумма"); ТЗ.Сортировать("Сумма Убыв");
ТЗ.НайтиСтроки(Новый Структура("Товар", Товар));

// Перенос одноимённых свойств
ЗаполнитьЗначенияСвойств(Приемник, Источник, "Товар,Количество,Цена");
```

## Объекты базы (сервер)

```1C
// Справочник
Эл = Справочники.Номенклатура.СоздатьЭлемент(); Эл.Наименование = "Чайник"; Эл.Записать();
Ссылка = Справочники.Номенклатура.НайтиПоКоду("000000001");
Объект = Ссылка.ПолучитьОбъект(); Объект.Наименование = "…"; Объект.Записать();

// Документ
Док = Документы.РасходнаяНакладная.СоздатьДокумент();
Док.Дата = ТекущаяДатаСеанса();
Стр = Док.Товары.Добавить();
Док.Записать(РежимЗаписиДокумента.Проведение);

// Константа, перечисление, предопределённое
Константы.НазваниеОрганизации.Получить();
Перечисления.ВидыНоменклатуры.Товар;
Справочники.Склады.ОсновнойСклад;
ПредопределенноеЗначение("Справочник.Склады.ОсновнойСклад");   // на клиенте

// Регистр сведений
Цена = РегистрыСведений.ЦеныНоменклатуры.ПолучитьПоследнее(Дата, Новый Структура("Номенклатура", Товар)).Цена;
```

## Запрос

```1C
Запрос = Новый Запрос;
Запрос.Текст =
"ВЫБРАТЬ
|	Т.Товар КАК Товар,
|	СУММА(Т.Сумма) КАК Сумма
|ИЗ
|	Документ.РасходнаяНакладная.Товары КАК Т
|ГДЕ
|	Т.Ссылка.Дата МЕЖДУ &Начало И &Конец
|	И Т.Ссылка.Проведен
|СГРУППИРОВАТЬ ПО
|	Т.Товар";
Запрос.УстановитьПараметр("Начало", НачалоМесяца(ТекущаяДатаСеанса()));
Запрос.УстановитьПараметр("Конец", КонецМесяца(ТекущаяДатаСеанса()));

Выборка = Запрос.Выполнить().Выбрать();
Пока Выборка.Следующий() Цикл
	// Выборка.Товар, Выборка.Сумма
КонецЦикла;
// или: Таблица = Запрос.Выполнить().Выгрузить();
```

## Сообщения и транзакции

```1C
Сообщение = Новый СообщениеПользователю;
Сообщение.Текст = "Не выбран покупатель";
Сообщение.Поле = "Покупатель";
Сообщение.УстановитьДанные(ЭтотОбъект);   // в модуле объекта
Сообщение.Сообщить();

НачатьТранзакцию();
Попытка
	// изменения
	ЗафиксироватьТранзакцию();
Исключение
	ОтменитьТранзакцию();
	ВызватьИсключение;
КонецПопытки;
```
